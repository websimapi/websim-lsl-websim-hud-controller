// Enhanced LSL WebSim HUD Script // Place this script in your HUD object integer LISTEN_CHANNEL = 0; integer listenHandle; string currentUrl = "https://websim.com/@api/lsl-websim-hud-controller"; integer messageCount = 0; float lastHeartbeat = 0.0; integer isConnected = FALSE; // Configuration float HEARTBEAT_TIMEOUT = 30.0; // seconds integer DEBUG_MODE = TRUE; handleUrlMessage(string msg) { currentUrl = msg; messageCount++; isConnected = TRUE; lastHeartbeat = llGetTime(); if (DEBUG_MODE) { llOwnerSay("URL Updated: " + currentUrl); } // You can add URL validation here if (llSubStringIndex(currentUrl, "websim") != -1) { llOwnerSay("WebSim URL detected: " + extractDomain(currentUrl)); } } // Handle heartbeat messages handleHeartbeat(string msg) { lastHeartbeat = llGetTime(); isConnected = TRUE; if (DEBUG_MODE) { llOwnerSay("Heartbeat received"); } } // Handle test messages handleTestMessage(string msg) { messageCount++; llOwnerSay("Test message received: " + msg); // Echo back a confirmation // Note: This would require a way to send data back to the web interface // For now, we just log it } // Handle status requests handleStatusRequest(string msg) { reportStatus(); } // Report current status reportStatus() { llOwnerSay("=== HUD Status Report ==="); llOwnerSay("Current URL: " + currentUrl); llOwnerSay("Messages received: " + (string)messageCount); if (isConnected) { llOwnerSay("Connection: Active"); } else { llOwnerSay("Connection: Inactive"); } llOwnerSay("Last heartbeat: " + (string)(llGetTime() - lastHeartbeat) + "s ago"); llOwnerSay("Debug mode: " + (string)DEBUG_MODE); } // Update HUD display updateDisplay() { string status; if (isConnected) { status = "Connected"; } else { status = "Disconnected"; } string displayText = "WebSim HUD\nStatus: " + status + "\nMessages: " + (string)messageCount; vector color; if (isConnected) { color = <0,1,0>; } else { color = <1,0,0>; } llSetText(displayText, color, 1.0); } // Extract domain from URL for cleaner display string extractDomain(string url) { integer start = llSubStringIndex(url, "://"); if (start != -1) { start += 3; integer end = llSubStringIndex(llGetSubString(url, start, -1), "/"); if (end != -1) { return llGetSubString(url, start, start + end - 1); } return llGetSubString(url, start, -1); } return url; } default { state_entry() { // Initialize the HUD llOwnerSay("WebSim HUD initialized and ready."); // Set up listening for chat commands (optional) listenHandle = llListen(LISTEN_CHANNEL, "", llGetOwner(), ""); // Set initial text if HUD has text capability llSetText("WebSim HUD\nStatus: Waiting\nMessages: 0", <1,1,1>, 1.0); // Set a timer for heartbeat monitoring llSetTimerEvent(5.0); } link_message(integer sender, integer num, string msg, key id) { // Parse incoming messages from web interface if (llSubStringIndex(msg, "http") == 0) { handleUrlMessage(msg); } else if (llSubStringIndex(msg, "HEARTBEAT") == 0) { handleHeartbeat(msg); } else if (llSubStringIndex(msg, "TEST") == 0) { handleTestMessage(msg); } else if (llSubStringIndex(msg, "STATUS") == 0) { handleStatusRequest(msg); } else { // Handle unknown message types if (DEBUG_MODE) { llOwnerSay("Unknown message received: " + msg); } } updateDisplay(); } listen(integer channel, string name, key id, string message) { // Handle chat commands from owner if (id == llGetOwner()) { if (message == "/hud status") { reportStatus(); } else if (message == "/hud reset") { llOwnerSay("Resetting HUD..."); llResetScript(); } else if (message == "/hud debug") { DEBUG_MODE = !DEBUG_MODE; llOwnerSay("Debug mode: " + (string)DEBUG_MODE); } } } timer() { // Check for heartbeat timeout float currentTime = llGetTime(); if (isConnected && (currentTime - lastHeartbeat) > HEARTBEAT_TIMEOUT) { isConnected = FALSE; llOwnerSay("Connection timeout - no heartbeat received"); updateDisplay(); } } }
Setup Instructions
IMPORTANT: LSL Bridge Setup Instructions 1. Enable Media on a Prim in Second Life: - Right-click on your HUD object - Select "Edit" - Go to "Media" tab - Check "Enable Media" - Set Media Type to "Web Page" - Set Home URL to: https://websim.com/@api/lsl-websim-hud-controller - Check "Auto Play Media" - Check "Auto Scale Media" 2. Enable LSL Bridge in Your Viewer: - In Firestorm: Preferences > Firestorm > Bridge > Enable LSL Bridge - In other viewers: Look for LSL Bridge or External Script options 3. Allow Script to Access Internet: - Make sure your land/parcel allows scripts - Ensure HTTP requests are enabled for your object 4. Alternative Setup - Use HTTP Requests: - If LSL Bridge doesn't work, the script can be modified to use llHTTPRequest() - Contact your HUD creator for HTTP-based version 5. Testing: - Copy the LSL script above into your HUD - Visit this page with media enabled on the HUD - You should see URL updates in local chat Note: LSL Bridge requires specific viewer support and may not work in all environments.
